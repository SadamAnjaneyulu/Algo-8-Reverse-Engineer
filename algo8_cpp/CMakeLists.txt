cmake_minimum_required(VERSION 3.10)
project(algo8 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# No fast-math, no FMA contraction: results must match the Python reference to the last digit.
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wno-misleading-indentation -ffp-contract=off")
find_package(Threads REQUIRED)
add_library(algo8 STATIC src/d1_engine.cpp src/d0_engine.cpp src/backtest.cpp)
target_include_directories(algo8 PUBLIC include)
add_executable(algo8_backtest src/main_backtest.cpp)
target_link_libraries(algo8_backtest algo8 Threads::Threads)
add_executable(dump_cycle tools/dump_cycle.cpp)
target_link_libraries(dump_cycle algo8 Threads::Threads)
add_executable(test_util tests/test_util.cpp)
target_include_directories(test_util PUBLIC include)
enable_testing()
add_test(NAME util COMMAND test_util)
add_test(NAME parity COMMAND ${CMAKE_COMMAND} -E env bash ${CMAKE_SOURCE_DIR}/tests/run_parity.sh ${CMAKE_BINARY_DIR})
