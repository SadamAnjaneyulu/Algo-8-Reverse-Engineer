cmake_minimum_required(VERSION 3.16)
project(portfolio_engine_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# -ffp-contract=off: no fused multiply-add, so AVG-method floating point follows the same operation order as the Python reference
add_compile_options(-Wall -Wextra -Wpedantic -ffp-contract=off)
include_directories(include)
add_executable(pe_replay src/pe_replay.cpp)
add_executable(pe_core_tests tests/test_core.cpp)
enable_testing()
add_test(NAME pe_core_tests COMMAND pe_core_tests)
