cmake_minimum_required(VERSION 3.16)
project(portfolio_construction_skeleton LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -ffp-contract=off)
# The unchanged M2 headers (pe/*.hpp) are found in the delivery layout or in the source-tree layout; override with -DM2_CPP_INCLUDE=<dir>
set(M2_CPP_INCLUDE "" CACHE PATH "directory containing pe/ledger.hpp")
if(NOT M2_CPP_INCLUDE AND DEFINED ENV{ALGO8_M2_CPP_INCLUDE})
  set(M2_CPP_INCLUDE "$ENV{ALGO8_M2_CPP_INCLUDE}")
endif()
if(NOT M2_CPP_INCLUDE)
  foreach(c "${CMAKE_CURRENT_SOURCE_DIR}/../../01_M2_Position_Reconstruction/01_CODE/cpp/include" "${CMAKE_CURRENT_SOURCE_DIR}/../../cpp/include")
    if(EXISTS "${c}/pe/ledger.hpp")
      set(M2_CPP_INCLUDE "${c}")
      break()
    endif()
  endforeach()
endif()
if(NOT M2_CPP_INCLUDE)
  message(FATAL_ERROR "M2 C++ headers not found; pass -DM2_CPP_INCLUDE=<dir that contains pe/ledger.hpp>")
endif()
include_directories(include ${M2_CPP_INCLUDE})
add_executable(pc_run src/pc_run.cpp)
add_executable(pc_tests tests/test_pc.cpp)
enable_testing()
add_test(NAME pc_tests COMMAND pc_tests)
